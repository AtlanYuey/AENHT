if [ "${{ steps.release.outputs.signed }}" != "true" ]; then
  echo "::error::No signing secrets: a release isn't given a debug-signed APK."
  exit 1
fi
VERSION=$(grep -oP 'versionName = "\K[^"]+' androidApp/build.gradle.kts)
cp dist/AENHT.apk "dist/AENHT-${VERSION}.apk"
uploaded=false
for i in 1 2 3; do
  if gh release upload "${{ github.event.release.tag_name }}" "dist/AENHT-${VERSION}.apk" --clobber; then uploaded=true; break; fi
  if [ "$i" -lt 3 ]; then sleep 10; fi
done
if [ "$uploaded" != true ]; then echo "::error::APK upload failed after three attempts"; exit 1; fi

NOTES="Newest development build from the dev branch (commit ${GITHUB_SHA::7}). Experimental: not a release."
gh release view dev-latest >/dev/null 2>&1 || \
  gh release create dev-latest --prerelease --target dev --title "Development build" --notes "$NOTES" || true
uploaded=false
for i in 1 2 3; do
  if gh release upload dev-latest iosApp/build/AENHT.ipa --clobber; then uploaded=true; break; fi
  if [ "$i" -lt 3 ]; then sleep 10; fi
done
if [ "$uploaded" != true ]; then echo "::error::IPA upload failed after three attempts"; exit 1; fi
gh release edit dev-latest --notes "$NOTES" || true
